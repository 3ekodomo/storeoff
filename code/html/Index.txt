    <div class="modal-overlay" id="jsonModal">
        <div class="modal-content">
            <div>
                <h3 style="margin:0; font-size: 16px; color: var(--accent);">Data Backup / Restore</h3>
                <p style="margin:5px 0 0 0; font-size: 12px; color: #aaa;">Copy text to backup, or paste valid JSON text to restore.</p>
            </div>
            <textarea id="jsonTextarea" placeholder="Paste JSON data here..."></textarea>
            <div class="modal-actions">
                <button onclick="copyJsonText()">Copy to Clipboard</button>

    <script>
        const canvas = document.getElementById('receiptCanvas');
        const ctx = canvas.getContext('2d');
        const backgroundImg = new Image();
        const refImg = new Image();
        const stageContainer = document.getElementById('stage');
        
        let bgImgUrl = 'https://raw.githubusercontent.com/3ekodomo/storeoff/main/2nd/rn/1000405411.jpg';
        backgroundImg.crossOrigin = 'Anonymous';
        
        refImg.crossOrigin = 'Anonymous';
        refImg.src = 'https://cdn.jsdelivr.net/gh/3ekodomo/storeoff@main/Temp/1785313218632_Screenshot_2026-07-29-08-16-13-094_com.phonepe.app.jpg';

        const fontFamily = "'Roboto', 'Segoe UI', Arial, sans-serif";
        let zoomLevel = 0.4; 
        
        function getCurrentFormattedDate() {
            const now = new Date();
            const months = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
            
            const day = now.getDate();
            const month = months[now.getMonth()];
            const year = now.getFullYear();
            
            let hours = now.getHours();
            const minutes = now.getMinutes().toString().padStart(2, '0');
            const ampm = hours >= 12 ? 'PM' : 'AM';
            
            hours = hours % 12;
            hours = hours ? hours : 12;
            const hoursStr = hours.toString().padStart(2, '0');
            
            return `${day} ${month} ${year} at ${hoursStr}:${minutes} ${ampm}`;
        }

        function getInitials(name) {
            const words = name.trim().split(/\s+/);
            if (words.length === 0 || words[0] === "") return "";
            if (words.length === 1) return words[0].charAt(0).toUpperCase();
            return (words[0].charAt(0) + words[words.length - 1].charAt(0)).toUpperCase();
        }

        const defaultNameStr = 'Suyash Sanjay Mhatre';
        
        const defaultElements = {
            date: { label: 'Date', text: getCurrentFormattedDate(), size: 36, x: 540, y: 495, font: '500', color: '#c5e89d', align: 'center', base: 'alphabetic' },
            initials: { label: 'Icon', text: getInitials(defaultNameStr), size: 55, x: 155, y: 622, font: '300', color: '#ffffff', align: 'center', base: 'middle' },
            name: { label: 'Name', text: defaultNameStr, size: 50, x: 280, y: 590, font: '700', color: '#ffffff', align: 'left', base: 'alphabetic' },
            upi: { label: 'UPI', text: 'suyashmhatre23@slc', size: 36, x: 280, y: 642, font: '400', color: '#9e9e9e', align: 'left', base: 'alphabetic' },
            amount: { label: 'Amt', text: '₹1000', size: 75, x: 280, y: 750, font: '700', color: '#ffffff', align: 'left', base: 'alphabetic' }
        };

        let textElements = JSON.parse(JSON.stringify(defaultElements));
        let activeKey = null;

        // --- Backup Export / Import via Text Modal ---
        const jsonModal = document.getElementById('jsonModal');
        const jsonTextarea = document.getElementById('jsonTextarea');

        function exportJSON() {
            const backupData = {
                bgImgUrl: bgImgUrl,
                textElements: textElements
            };
            jsonTextarea.value = JSON.stringify(backupData, null, 2);
            jsonModal.style.display = 'flex';
        }

        function importJSON() {
            jsonTextarea.value = "";
            jsonModal.style.display = 'flex';
            jsonTextarea.focus();
        }

        function closeJsonModal() {
            jsonModal.style.display = 'none';
        }

        function copyJsonText() {
            jsonTextarea.select();
            document.execCommand('copy');
            alert("JSON copied to clipboard!");
        }

        function applyJsonText() {
            try {
                const rawText = jsonTextarea.value.trim();
                if (!rawText) {
                    alert("Please paste valid JSON data first.");
                    return;
                }
                const parsedData = JSON.parse(rawText);
                
                if (parsedData.textElements) {
                    for (let key in textElements) {
                        if (parsedData.textElements[key]) {
                            textElements[key] = { ...textElements[key], ...parsedData.textElements[key] };
                        }
                    }
                }
                
                if (parsedData.bgImgUrl) {
                    bgImgUrl = parsedData.bgImgUrl;
                    backgroundImg.src = bgImgUrl; 
                }

                saveConfig();
                activeKey = null;
                updateInspectorUI();
                draw();
                closeJsonModal();
                alert("Data imported successfully!");
                
            } catch (error) {
                alert("Invalid JSON data. Please make sure you copied it correctly.");
            }
        }

        // --- Core Functions ---
        function loadConfig() {
            const savedData = localStorage.getItem('receiptData');
            const savedBg = localStorage.getItem('receiptBgUrl');
            
            if (savedBg) {
                bgImgUrl = savedBg;
            }
            
            if (savedData) {
                try {
                    const parsed = JSON.parse(savedData);
                    for (let key in textElements) {
                        if (parsed[key]) textElements[key] = { ...textElements[key], ...parsed[key] };
                    }
                } catch(e) {}
            }
            
            backgroundImg.src = bgImgUrl;
        }

        function saveConfig() {
            localStorage.setItem('receiptData', JSON.stringify(textElements));
            localStorage.setItem('receiptBgUrl', bgImgUrl);
        }

        function resetConfig() {
            if(confirm("Erase saved config and reset to defaults?")) {
                localStorage.removeItem('receiptData');
                localStorage.removeItem('receiptBgUrl');
                defaultElements.date.text = getCurrentFormattedDate();
                textElements = JSON.parse(JSON.stringify(defaultElements));
                bgImgUrl = 'https://raw.githubusercontent.com/3ekodomo/storeoff/main/2nd/rn/1000405411.jpg';
                backgroundImg.src = bgImgUrl;
                activeKey = null;
                updateInspectorUI();
            }
        }
        
        function changeBackgroundImage() {
            const newUrl = prompt("Enter new background image URL:", bgImgUrl);
            if (newUrl && newUrl.trim() !== "") {
                bgImgUrl = newUrl.trim();
                backgroundImg.src = bgImgUrl;
                saveConfig();
            }
        }

        function applyZoom() {
            if (!backgroundImg.complete) return;
            document.getElementById('zoom_val').innerText = Math.round(zoomLevel * 100) + '%';
            canvas.style.width = (backgroundImg.naturalWidth * zoomLevel) + 'px';
            canvas.style.height = (backgroundImg.naturalHeight * zoomLevel) + 'px';
        }

        function changeZoom(delta) {
            zoomLevel += delta;
            if(zoomLevel < 0.1) zoomLevel = 0.1;
            if(zoomLevel > 3.0) zoomLevel = 3.0;
            applyZoom();
        }

        function updateInspectorUI() {
            const controls = document.getElementById('editControls');
            const centerBtn = document.getElementById('centerBtn');
            const syncDateBtn = document.getElementById('syncDateBtn');
            
            if (activeKey && textElements[activeKey]) {
                const el = textElements[activeKey];
                controls.classList.remove('disabled-group');
                document.getElementById('prop_title').innerText = el.label;
                document.getElementById('prop_text').value = el.text;
                document.getElementById('prop_size').value = el.size;
                document.getElementById('prop_weight').value = el.font;
                document.getElementById('prop_color').value = el.color;
                
                if (activeKey === 'date') {
                    centerBtn.style.display = 'inline-block';
                    syncDateBtn.style.display = 'inline-block';
                } else {
                    centerBtn.style.display = 'none';
                    syncDateBtn.style.display = 'none';
                }
            } else {
                controls.classList.add('disabled-group');
                document.getElementById('prop_title').innerText = "None";
                document.getElementById('prop_text').value = "";
                document.getElementById('prop_size').value = "";
                document.getElementById('prop_color').value = "";
                centerBtn.style.display = 'none';
                syncDateBtn.style.display = 'none';
            }
        }

        function updateElement(prop) {
            if (!activeKey) return;
            const el = textElements[activeKey];
            
            if (prop === 'text') {
                el.text = document.getElementById('prop_text').value;
                if (activeKey === 'name') {
                    textElements['initials'].text = getInitials(el.text);
                }
            }
            if (prop === 'size') el.size = parseInt(document.getElementById('prop_size').value) || 10;
            if (prop === 'font') el.font = document.getElementById('prop_weight').value;
            if (prop === 'color') el.color = document.getElementById('prop_color').value;
            
            saveConfig(); 
            draw();
        }

        function centerDate() {
            if (activeKey === 'date' && backgroundImg.complete) {
                textElements['date'].x = backgroundImg.naturalWidth / 2;
                saveConfig();
                draw();
            }
        }

        function syncDate() {
            if (activeKey === 'date') {
                const newDate = getCurrentFormattedDate();
                textElements['date'].text = newDate;
                document.getElementById('prop_text').value = newDate;
                saveConfig();
                draw();
            }
        }

        backgroundImg.onload = () => {
            canvas.width = backgroundImg.naturalWidth;
            canvas.height = backgroundImg.naturalHeight;
            applyZoom();
            draw();
        };

        loadConfig();
        
        document.fonts.ready.then(() => {
            if(backgroundImg.complete) {
                canvas.width = backgroundImg.naturalWidth;
                canvas.height = backgroundImg.naturalHeight;
                applyZoom();
                draw();
            }
        });

        function draw() {
            if (!backgroundImg.complete) return;
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(backgroundImg, 0, 0, canvas.width, canvas.height);

            for (const [key, data] of Object.entries(textElements)) {
                ctx.font = `${data.font} ${data.size}px ${fontFamily}`;
                ctx.fillStyle = data.color;
                ctx.textAlign = data.align;
                ctx.textBaseline = data.base;
                ctx.fillText(data.text, data.x, data.y);

                if (key === activeKey) {
                    const metrics = ctx.measureText(data.text);
                    const width = metrics.width;
                    const height = data.size; 
                    const pad = 10;
                    let left = data.x;
                    if (data.align === 'center') left = data.x - (width / 2);
                    else if (data.align === 'right') left = data.x - width;
                    let top = data.y - height;
                    if (data.base === 'middle') top = data.y - (height / 2);
                    
                    ctx.save();
                    ctx.strokeStyle = '#bb86fc';
                    ctx.lineWidth = 3;
                    ctx.setLineDash([10, 10]);
                    ctx.strokeRect(left - pad, top - pad, width + (pad*2), height + (pad*2));
                    ctx.restore();
                }
            }

            if (refImg.complete && document.getElementById("overlay_show").checked) {
                const opacity = document.getElementById("overlay_opacity").value;
                ctx.save();
                ctx.globalAlpha = opacity / 100;
                ctx.drawImage(refImg, 0, 0, canvas.width, canvas.height);
                ctx.restore();
            }
        }

        // --- Interaction Logic ---
        let isDraggingText = false;
        let isPanningCanvas = false;
        let isPinching = false;
        
        let dragOffsetX = 0;
        let dragOffsetY = 0;
        
        let startClientX = 0;
        let startClientY = 0;
        let startScrollLeft = 0;
        let startScrollTop = 0;
        
        let initialPinchDistance = null;
        let initialZoomLevel = 1;

        function getDistance(touch1, touch2) {
            return Math.hypot(touch2.clientX - touch1.clientX, touch2.clientY - touch1.clientY);
        }

        function getCenter(touches) {
            return {
                x: (touches[0].clientX + touches[1].clientX) / 2,
                y: (touches[0].clientY + touches[1].clientY) / 2
            };
        }

        function getMousePos(evt) {
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            let clientX = evt.clientX;
            let clientY = evt.clientY;
            if (evt.touches && evt.touches.length > 0) {
                clientX = evt.touches[0].clientX;
                clientY = evt.touches[0].clientY;
            }
            return {
                x: (clientX - rect.left) * scaleX,
                y: (clientY - rect.top) * scaleY
            };
        }

        function hitTest(x, y) {
            for (const [key, data] of Object.entries(textElements)) {
                ctx.font = `${data.font} ${data.size}px ${fontFamily}`;
                const metrics = ctx.measureText(data.text);
                const width = metrics.width;
                const height = data.size; 
                const pad = 30; 
                
                let left = data.x;
                if (data.align === 'center') left = data.x - (width / 2);
                let top = data.y - height;
                if (data.base === 'middle') top = data.y - (height / 2);

                if (x >= left - pad && x <= left + width + pad && y >= top - pad && y <= top + height + pad) {
                    return key;
                }
            }
            return null;
        }

        function onStart(e) {
            if (e.target.tagName === 'BUTTON' || e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') {
                return; // Do not trigger canvas interactions if clicking a UI element
            }
            
            if (e.touches && e.touches.length === 2) {
                e.preventDefault();
                isPinching = true;
                initialPinchDistance = getDistance(e.touches[0], e.touches[1]);
                initialZoomLevel = zoomLevel;
                
                const center = getCenter(e.touches);
                startClientX = center.x;
                startClientY = center.y;
                startScrollLeft = stageContainer.scrollLeft;
                startScrollTop = stageContainer.scrollTop;

                isDraggingText = false;
                isPanningCanvas = false;
                return;
            }

            let clientX = e.clientX || (e.touches && e.touches[0].clientX);
            let clientY = e.clientY || (e.touches && e.touches[0].clientY);
            
            if(e.target === canvas) {
                const pos = getMousePos(e);
                const clickedElement = hitTest(pos.x, pos.y);
                
                if (clickedElement) {
                    e.preventDefault();
                    activeKey = clickedElement;
                    isDraggingText = true;
                    updateInspectorUI();
                    
                    const el = textElements[activeKey];
                    dragOffsetX = pos.x - el.x;
                    dragOffsetY = pos.y - el.y;
                    draw();
                    return; 
                } 
            }
            
            activeKey = null;
            updateInspectorUI();
            draw();
            
            isPanningCanvas = true;
            startClientX = clientX;
            startClientY = clientY;
            startScrollLeft = stageContainer.scrollLeft;
            startScrollTop = stageContainer.scrollTop;
        }

        function onMove(e) {
            if (isPinching && e.touches && e.touches.length === 2) {
                e.preventDefault();
                
                const currentDistance = getDistance(e.touches[0], e.touches[1]);
                const ratio = currentDistance / initialPinchDistance;
                zoomLevel = initialZoomLevel * ratio;
                if(zoomLevel < 0.1) zoomLevel = 0.1;
                if(zoomLevel > 3.0) zoomLevel = 3.0;
                applyZoom();

                const center = getCenter(e.touches);
                stageContainer.scrollLeft = startScrollLeft - (center.x - startClientX);
                stageContainer.scrollTop = startScrollTop - (center.y - startClientY);
                return;
            }

            let clientX = e.clientX || (e.touches && e.touches[0].clientX);
            let clientY = e.clientY || (e.touches && e.touches[0].clientY);

            if (isDraggingText && activeKey) {
                e.preventDefault();
                const pos = getMousePos(e);
                textElements[activeKey].x = pos.x - dragOffsetX;
                textElements[activeKey].y = pos.y - dragOffsetY;
                draw();
            } else if (isPanningCanvas) {
                e.preventDefault(); 
                stageContainer.scrollLeft = startScrollLeft - (clientX - startClientX);
                stageContainer.scrollTop = startScrollTop - (clientY - startClientY);
            }
        }

        function onEnd(e) {
            if (isPinching) {
                if (!e.touches || e.touches.length < 2) {
                    isPinching = false;
                    initialPinchDistance = null;
                }
                return;
            }

            if (isDraggingText) saveConfig(); 
            isDraggingText = false;
            isPanningCanvas = false;
        }

        stageContainer.addEventListener('mousedown', onStart);
        window.addEventListener('mousemove', onMove);
        window.addEventListener('mouseup', onEnd);
        
        stageContainer.addEventListener('touchstart', onStart, { passive: false });
        window.addEventListener('touchmove', onMove, { passive: false });
        window.addEventListener('touchend', onEnd);

        function downloadImage() {
            const previousActive = activeKey;
            activeKey = null;
            const showOverlay = document.getElementById("overlay_show").checked;
            document.getElementById("overlay_show").checked = false;
            draw();

            const link = document.createElement('a');
            link.download = 'edited_receipt_final.jpg';
            link.href = canvas.toDataURL('image/jpeg', 1.0);
            link.click();

            activeKey = previousActive;
            document.getElementById("overlay_show").checked = showOverlay;
            draw();
        }
    </script>
